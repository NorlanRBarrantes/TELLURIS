Evaluación Metodológica: ¿Muestra Puntual ($1 \times 1$) o Ventana Multi-Píxel ($3 \times 3$ / $5 \times 5$)?3.1 Por qué una ventana multi-píxel ($3 \times 3$ de $750\text{ m} \times 750\text{ m}$) es científicamente muy superior a un único píxel ($1 \times 1$ de $250\text{ m}$)Consultar kmAboveBelow=0&kmLeftRight=0 ($1 \times 1\text{ píxel}$) en MOD13Q1 tiene tres vulnerabilidades conocidas en la literatura de validación de MODIS (Tan et al., 2006; Wolfe et al., 2002):Incertidumbre de geolocalización y Point Spread Function (PSF) triangular de MODIS: La incertidumbre de geolocalización de MODIS Terra ($1\sigma \approx 50\text{ m}$) y el efecto de barrido fuera del nadir (bow-tie) hacen que el valor asignado a un píxel individual de $250\text{ m}$ provenga en realidad de una huella elíptica que se solapa con los píxeles vecinos.Perturbaciones estocásticas micro-locales: La caída de un solo árbol emergente por viento, un derrumbe puntual en la montaña o una pequeña finca de $3\text{ ha}$ altera un píxel de $250\text{ m}$, pero no representa una tendencia climática regional.Pérdida severa de muestras por nubes aisladas (Trade-wind cumuli): En el Caribe de Costa Rica, pequeños cúmulos de los vientos alisios tapan el píxel central (4, 4) en una fecha dada, dejando despejados los píxeles contiguos a $250\text{ m}$ de distancia. Con una muestra puntual ($1 \times 1$), esa fecha se pierde como NaN; con una ventana de $3 \times 3\text{ píxeles}$ ($9\text{ píxeles} = 0.75\text{ km} \times 0.75\text{ km}$), se calcula la mediana espacial de los píxeles libres de nubes, recuperando compuestos válidos y reduciendo la varianza de ruido radiométrico en un factor de $\sim \frac{1}{\sqrt{m}}$.3.2 Cómo Configurar la Extracción Multi-Píxel en la API REST de ORNL DAAC (TESViS)El endpoint /subset de ORNL DAAC soporta nativamente ventanas espaciales alrededor de la coordenada central sin requerir más llamadas HTTP:Para una ventana de $1 \times 1\text{ píxel}$ ($250\text{ m} \times 250\text{ m}$): kmAboveBelow=0&kmLeftRight=0 (devuelve un vector de 1 elemento por fecha).Para una ventana de $5 \times 5\text{ píxeles}$ ($1.25\text{ km} \times 1.25\text{ km}$): kmAboveBelow=0.5&kmLeftRight=0.5 (o kmAboveBelow=1&kmLeftRight=1 para $9 \times 9\text{ píxeles}$, de la cual se recorta la sub-matriz central de $3 \times 3\text{ píxeles}$ índices [3:6, 3:6]).4. def decode_mod13_vi_quality(qa_val: int) -> Dict[str, int]:
    """Extrae los campos de bits oficiales de '250m_16_days_VI_Quality' (uint16)."""
    qa = int(qa_val) & 0xFFFF
    return {
        "modland_qa": qa & 0b11,                 # Bits 0-1
        "vi_usefulness": (qa >> 2) & 0b1111,     # Bits 2-5
        "aerosol_quantity": (qa >> 6) & 0b11,    # Bits 6-7
        "adjacent_cloud": (qa >> 8) & 0b1,       # Bit 8
        "brdf_coupled": (qa >> 9) & 0b1,         # Bit 9
        "mixed_clouds": (qa >> 10) & 0b1,        # Bit 10
        "land_water_mask": (qa >> 11) & 0b111,   # Bits 11-13 (1 == Land)
        "possible_snow_ice": (qa >> 14) & 0b1,   # Bit 14
        "possible_shadow": (qa >> 15) & 0b1,     # Bit 15
    }


def is_clean_land_observation(qa_val: int) -> bool:
    """
    Criterio estricto para un píxel válido en tierra firme:
    - Bits 11-13 (land_water_mask) == 1 (Tierra firme inequívoca)
    - Bits 0-1 (modland_qa) in (0, 1)
    - Sin aerosoles altos (Bits 6-7 != 3), sin nube adyacente, sin nubes mixtas ni sombra.
    """
    b = decode_mod13_vi_quality(qa_val)
    return (
        b["land_water_mask"] == 1
        and b["modland_qa"] in (0, 1)
        and b["aerosol_quantity"] != 3
        and b["adjacent_cloud"] == 0
        and b["mixed_clouds"] == 0
        and b["possible_snow_ice"] == 0
        and b["possible_shadow"] == 0
    )


def check_merra2_cell_match(lat: float, lon: float, ref_lat: float = 10.0, ref_lon: float = -83.125) -> bool:
    """Verifica si la coordenada cae exactamente en la misma celda MERRA-2 (0.5° x 0.625°) que Limón."""
    return (abs(lat - ref_lat) <= 0.25) and (abs(lon - ref_lon) <= 0.3125)


def fetch_ornl_subset(
    product: str, band: str, lat: float, lon: float, start_date: str, end_date: str, km_radius: float = 0.0
) -> Dict[str, Any]:
    """Consulta el endpoint síncrono /subset de ORNL DAAC TESViS API v1."""
    url = (
        f"{ORNL_BASE_URL}/{product}/subset"
        f"?latitude={lat}&longitude={lon}&band={band}"
        f"&startDate={start_date}&endDate={end_date}"
        f"&kmAboveBelow={km_radius}&kmLeftRight={km_radius}"
    )
    headers = {"Accept": "application/json"}
    resp = requests.get(url, headers=headers, timeout=30)
    resp.raise_for_status()
    return resp.json()


def audit_site_spatial_qa(lat: float, lon: float, sample_start: str = "A2023001", sample_end: str = "A2023145") -> Dict[str, Any]:
    """
    Audita una ubicación consultando una ventana de 1x1 km (que contiene la matriz central 3x3 de 250m)
    en MOD13Q1 ('250m_16_days_VI_Quality') y la cobertura terrestre en MCD12Q1 ('LC_Type1').
    """
    # 1. Consultar VI_Quality en ventana con km_radius=1.0 para extraer píxel central y vecindad 3x3
    qa_payload = fetch_ornl_subset(
        "MOD13Q1", "250m_16_days_VI_Quality", lat, lon, sample_start, sample_end, km_radius=1.0
    )
    nrows = int(qa_payload.get("nrows", 1))
    ncols = int(qa_payload.get("ncols", 1))
    center_idx = (nrows // 2) * ncols + (ncols // 2)

    center_lw_counts: Dict[str, int] = {}
    single_pixel_valid_dates = 0
    window_3x3_valid_dates = 0
    total_dates = len(qa_payload.get("subset", []))

    for step in qa_payload.get("subset", []):
        data_flat: List[int] = step["data"]
        arr_2d = np.array(data_flat, dtype=np.int64).reshape((nrows, ncols))

        # Píxel central (1x1)
        r_c, c_c = nrows // 2, ncols // 2
        center_qa = int(arr_2d[r_c, c_c])
        lw_code = decode_mod13_vi_quality(center_qa)["land_water_mask"]
        lw_label = f"{lw_code} ({LAND_WATER_CODES.get(lw_code, 'Unknown')})"
        center_lw_counts[lw_label] = center_lw_counts.get(lw_label, 0) + 1

        if is_clean_land_observation(center_qa):
            single_pixel_valid_dates += 1

        # Sub-ventana 3x3 alrededor del centro
        r0, r1 = max(0, r_c - 1), min(nrows, r_c + 2)
        c0, c1 = max(0, c_c - 1), min(ncols, c_c + 2)
        sub_3x3 = arr_2d[r0:r1, c0:c1].flatten()
        clean_count_3x3 = sum(1 for val in sub_3x3 if is_clean_land_observation(int(val)))
        # Se acepta la fecha en la ventana 3x3 si al menos 3 de los 9 píxeles son tierra limpia
        if clean_count_3x3 >= 3:
            window_3x3_valid_dates += 1

    # 2. Consultar cobertura terrestre MCD12Q1 (LC_Type1 IGBP)
    lc_payload = fetch_ornl_subset("MCD12Q1", "LC_Type1", lat, lon, "A2022001", "A2023001", km_radius=0.0)
    lc_classes = []
    for step in lc_payload.get("subset", []):
        code = int(step["data"][0])
        lc_classes.append(f"{code}: {IGBP_CLASSES.get(code, 'Unknown')}")

    passes_land_mask = all(k.startswith("1 ") for k in center_lw_counts.keys()) and len(center_lw_counts) > 0

    return {
        "coordinates": {"lat": lat, "lon": lon},
        "same_merra2_cell_as_limon_port": check_merra2_cell_match(lat, lon),
        "mod13q1_center_pixel_land_water_distribution": center_lw_counts,
        "mcd12q1_igbp_land_cover": lc_classes,
        "sample_composites_evaluated": total_dates,
        "valid_clean_dates_single_pixel_1x1": single_pixel_valid_dates,
        "valid_clean_dates_window_3x3": window_3x3_valid_dates,
        "passes_unequivocal_land_criterion": passes_land_mask,
    }

def test_land_water_bit_extraction_matches_limon_finding():
    """
    Verifica que los bits 11-13 de VI_Quality identifiquen correctamente:
    - land_water_mask == 2 (Shoreline/Coastline) y 0 (Shallow ocean) como INVÁLIDOS.
    - land_water_mask == 1 (Land) limpio como VÁLIDO.
    """
    # Caso 1: Shoreline ( código 2 en bits 11-13 -> 2 << 11 = 4096 )
    qa_shoreline = (2 << 11) | 0b00
    assert decode_mod13_vi_quality(qa_shoreline)["land_water_mask"] == 2
    assert is_clean_land_observation(qa_shoreline) is False

    # Caso 2: Shallow Ocean ( código 0 en bits 11-13 )
    qa_ocean = (0 << 11) | 0b00
    assert decode_mod13_vi_quality(qa_ocean)["land_water_mask"] == 0
    assert is_clean_land_observation(qa_ocean) is False

    # Caso 3: Tierra inequívoca ( código 1 en bits 11-13 -> 1 << 11 = 2048 ), aerosol bajo (1 << 6)
    qa_clean_land = (1 << 11) | (1 << 6) | 0b00
    assert decode_mod13_vi_quality(qa_clean_land)["land_water_mask"] == 1
    assert is_clean_land_observation(qa_clean_land) is True
